REQUIRED SUPABASE SQL SETUP

Execute this script inside your Supabase project's SQL Editor to set up the mapping schema and allow public overrides: 

-- Create table for overrides and virtual custom SKUs
create table if not exists sku_images (
  sku text primary key,
  image_url text not null,       -- Base64 encoded file data
  title text,                    -- Title string override
  price text,                    -- Retail price string
  description text,              -- Overridden specs or instructions
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Enable default Row Level Security
alter table sku_images enable row level security;

-- Configure global access rights mapping policies
create policy "Allow general select" on sku_images for select using (true);
create policy "Allow general insert" on sku_images for insert with check (true);
create policy "Allow general update" on sku_images for update using (true);
